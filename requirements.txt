# Internets IRC bot - full runtime dependencies.
#
# Install everything the bot needs for full functionality:
#
#     pip install -r requirements.txt
#
# Each package is documented inline below.  Dev/test/CI deps (pytest,
# coverage, bandit, pip-audit, build, etc.) live in pyproject.toml
# under [project.optional-dependencies] dev - install with
# `pip install -e ".[dev]"` when working on the code itself.
#
# Version policy:  each lower bound is a security or compatibility FLOOR,
# not the latest release - requirements.lock routinely pins newer.  A
# floor moves when an advisory forces it (the ID is in the comment beside
# it) or when a functional reason does; nothing re-audits a floor once
# set, so it is a point-in-time judgment.  No upper bounds - pip pulls
# the newest compatible release, and CI on Linux, macOS, and Windows
# across Python 3.10 → 3.14 surfaces breakage from a future upstream
# release on the next push.  Raising a floor: docs/dependencies.md.
#
# requirements.lock is the hash-pinned resolution of this file and must
# be regenerated with scripts/regen-lockfile.sh on Python 3.10, never on
# a newer interpreter - resolving higher drops the transitives that are
# marker-gated to older Pythons.  See docs/known-issues.md item 6.

# Core HTTP client.  >=2.32.3 carries the PoolManager / verify fix
# (CVE-2024-35195) and the proxy credential leak fix (CVE-2023-32681).
requests>=2.32.3

# Async HTTP client - preferred transport for weather provider calls;
# the bot falls back to requests-in-thread when missing.
# >=3.14.3 carries PYSEC-2026-3545/3546/3547
# (subsumes the >=3.14.1 request-smuggling / parser fixes).
aiohttp>=3.14.3

# Argon2id - recommended admin password hasher (memory-hard, GPU-resistant).
# Required for `password_hash = argon2$...` in config.local.ini [admin].
argon2-cffi>=23.1.0

# bcrypt - alternative admin password hasher.
# >=4.2.0 is the Rust-backed implementation with current wheels for
# every major platform.  If you're on an exotic platform without a wheel
# and without Rust, comment this out and use scrypt (in stdlib hashlib).
bcrypt>=4.2.0

# PyJWT + cryptography - Apple WeatherKit JWT signing (ES256).
# PyJWT >=2.13.0 carries the 2026 PYSEC fixes (subsumes the 2.10.1
# CVE-2024-53861 JOSE header-confusion floor).
# cryptography >=50.0.0 carries PYSEC-2026-3552
# (subsumes the >=48.0.1 GHSA-537c-gmf6-5ccf floor).
PyJWT>=2.13.0
cryptography>=50.0.0

# Hardened XML parser.  Blocks billion-laughs DoS in addition to stdlib's
# XXE protection.  Imported at module scope by modules/idlerpg.py,
# modules/scinews.py, and modules/reflookup.py, so all three fail to load
# without it.  reflookup parses XML only on the .arxiv path but imports at
# module scope, so its other seven commands go down with it.
defusedxml>=0.7.1
